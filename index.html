(unique / total) stability = 1 - repetition_pressure return round(stability, 3) # ========================================================= # 🪞 MIRRA (STRICT PRESENT RENDER — NO DRIFT) # ========================================================= def mirra(mapping): return "\n".join([f"{m['char']} → {m['node']}" for m in mapping]) # ========================================================= # 🔊 PRAX AUDIO ENGINE (NODE → TONE) # ========================================================= def audio_trace(mapping): return [ { "node": m["node"], "tone": m["freq"], "phase": (i * 0.12) % 1 } for i, m in enumerate(mapping) ] # ========================================================= # 🌐 3D LATTICE PROJECTION (DATA ONLY) # ========================================================= def lattice_position(node_str): # converts node into pseudo 3D coordinate f = node_str[0] x = int(node_str[1]) y = int(node_str[2]) z = int(node_str[3]) return { "field": f, "x": x, "y": y, "z": z } # ========================================================= # 🧠 SYSTEM EXECUTION PIPELINE # ========================================================= def execute(text): mapping = map_input(text) # GEMMA mem_size = gemma_log({ "time": time.time(), "input": text, "mapping": mapping }) # SYNARIS coherence = synaris(mapping) # MIRRA render = mirra(mapping) # AUDIO audio = audio_trace(mapping) # 3D LATTICE spatial = [lattice_position(m["node"]) for m in mapping] return { "mirra": render, "coherence": coherence, "gemma_size": mem_size, "audio": audio, "spatial": spatial } # ========================================================= # 🌐 SINGLE PAGE UI # ========================================================= PAGE = """ Praxipent System v2
